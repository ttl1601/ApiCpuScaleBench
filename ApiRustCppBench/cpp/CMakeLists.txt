cmake_minimum_required(VERSION 3.20)
project(json_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(json-api main.cpp)
target_include_directories(json-api PRIVATE /usr/include/jsoncpp)
target_link_libraries(json-api PRIVATE drogon trantor jsoncpp pthread nlohmann_json::nlohmann_json)
target_compile_options(json-api PRIVATE -O3 -flto)
target_link_options(json-api PRIVATE -flto)
